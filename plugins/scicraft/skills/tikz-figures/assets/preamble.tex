% =====================================================================
%  preamble.tex : one preamble for every TikZ / PGFPlots paper figure.
%
%  Copy this file verbatim as the top of each figure file and put the
%  figure body (one tikzpicture) before \end{document}.
%
%  THEME SWITCH (edit the two lines below, or set them from the shell:
%     pdflatex "\def\figtheme{serif}\input{fig.tex}")
%    \figtheme   minimal : sans text and math, 4-tone palette
%                          (dark blue, light blue, mustard gold, greys). DEFAULT.
%                serif   : Palatino text and math, black outlines,
%                          blue family plus one warm red contrast.
%    \figfamily  auto    : the theme's font (minimal = sans, serif = Palatino)
%                sans    : Helvetica clone + sans math (helvet + sansmath)
%                serif   : Palatino text + math (mathpazo)
%                cm      : TeX default Computer Modern, to match a venue
%                          paper whose body math is Computer Modern
%  Every color and style below has the SAME NAME in both themes, so a
%  figure body switches theme without edits.
%
%  Compile:  pdflatex -interaction=nonstopmode -halt-on-error fig.tex
%  Frame decks (animation): add multi=tikzpicture to the class options.
%  Needs: standalone, tikz, pgfplots, helvet + sansmath (sans), mathpazo
%  (serif). All are in a full TeX Live; see references/workflow_qa.md for
%  a minimal tree.
% =====================================================================

% Depth and repetition macros (written for this preamble). A copy at depth d
% sits d steps up and right of the front position and is drawn with opacity
% 1 - d/(n+1), deepest first, so nearer copies cover farther ones.
% \nnstack{(x,y)}{style}{w}{h}{n}{dx}{dy}: n empty copies of size w x h behind
% the position (x,y); draw the labeled front node yourself afterwards.
\newcommand{\nnstack}[7]{%
  \pgfmathtruncatemacro\figN{#5}%
  \foreach \dd in {\figN,...,1}{%
    \path #1 ++({\dd*(#6)},{\dd*(#7)})
      node[#2, minimum width=#3, minimum height=#4, opacity={1-\dd/(\figN+1)}] {};}}
% \StackBox{(x,y)}{style}{n}{label}: as \nnstack, sized by the label itself
% (copies carry the label invisibly), then the labeled front node.
\newcommand{\StackBox}[4]{%
  \pgfmathtruncatemacro\figN{#3}%
  \foreach \dd in {\figN,...,1}{%
    \path #1 ++({\dd*1.4ex},{\dd*1ex})
      node[#2, opacity={1-\dd/(\figN+1)}, text opacity=0] {#4};}
  \path #1 node[#2] {#4};}
% \opstack{(x,y)}{label}: a module shared by k inputs, as two faded operator
% boxes behind a labeled front box named `opfront`. Wire grey `agg` arrows from
% the back copies and one black `spine` arrow from the front.
\newcommand{\opstack}[2]{%
  \foreach \dd in {2,1}{\path #1 ++({\dd*1ex},{\dd*0.8ex}) node[opbox faint] {};}
  \path #1 node[opbox] (opfront) {#2};}
% \tokenrow{(x,y)}{style}{n}{gap}: n token cells in a row, the first at (x,y).
\newcommand{\tokenrow}[4]{%
  \pgfmathtruncatemacro\figN{#3-1}%
  \foreach \cc in {0,...,\figN}{\path #1 ++({\cc*(#4)},0) node[#2] {};}}
