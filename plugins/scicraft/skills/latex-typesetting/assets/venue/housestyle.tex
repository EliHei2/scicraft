% =====================================================================
%  housestyle.tex: house additions on top of an UNMODIFIED venue style
%  (ICLR, NeurIPS, ICML, a journal class).
%
%  The venue style file owns the page geometry, the type sizes, the
%  section heads, the title block, the abstract, the running head and the
%  bibliography style. Do not override any of it here: no geometry, no
%  font size, no \...geometryfalse switch. Reviewers and checkers read
%  such changes as a format violation.
%
%  This file adds only what the venue leaves open: one accent color for
%  links and callout rules, a monospace that sits beside Times, microtype,
%  float placement, caption form, display-math rhythm, table helpers,
%  callout boxes without tcolorbox, theorem environments and cleveref.
%
%  Load order in main.tex:
%     \usepackage{<venue style>}   (with `times' if the venue loads it so)
%     \input{housestyle}
%  Build with pdfLaTeX: pdflatex, bibtex, pdflatex until no "Rerun".
% =====================================================================

% ---- wide floats ----
% A figure designed wider than the measure spends margin instead of being
% scaled below its label size. The overhang goes to the RIGHT, because the
% left margin carries the submission line numbers:
% \widefig{7in}{...} on a 5.5in measure overhangs 0.15in left, 1.35in right.
\newcommand{\widefig}[2]{%
  \makebox[\linewidth][c]{\hspace*{1.2in}\begin{minipage}{#1}#2\end{minipage}}}

% ---- run-in paragraph head that costs no vertical space ----
\newcommand{\rwpara}[1]{\par\noindent\textbf{#1}\hspace{0.2em}}

% ---- table helpers ----
% Row-group heading: an italic label across the width, instead of a
% category column repeated down every row. \grp{<number of columns>}{label}
% (no optional argument: a row that starts with \addlinespace must expand to
% \noalign first, which a command with an optional argument does not).
\newcommand{\grp}[2]{\addlinespace[3pt]\multicolumn{#1}{@{}l}{\itshape #2}\\[1pt]}
% Best is \textbf; second is \snd; the pair is declared in the caption.
\newcommand{\snd}[1]{\underline{#1}}
% Error after a value, small and gray: .58\pmsd{.09}
\newcommand{\pmsd}[1]{{\tiny\color{black!62}$\pm$#1}}
% Small gray annotation inside a header cell (sample counts, arrows)
\newcommand{\hnote}[1]{{\tiny\color{black!62}#1}}
% p value with an upright, thin-spaced equals sign: \pv{.002}
\newcommand{\pv}[1]{\textit{p}\,=\,#1}
% Not applicable
\newcommand{\na}{\textcolor{soft}{--}}
% Ragged-right paragraph column: below about 40 characters, justification
% leaves word spaces the eye reads as holes.
\newcolumntype{R}[1]{>{\raggedright\arraybackslash}p{#1}}

% ---- callouts: square corners, hidden frame, accent left rule, pale fill ----
% Few, one style. Jobs that earned a box: the contributions block, and a
% one-sentence takeaway closing a dense section. Core LaTeX only, because
% an older kernel can refuse a newer tcolorbox.
\makeatletter
\newsavebox{\hs@cbox}
\newlength{\hs@rule}\setlength{\hs@rule}{1.8pt}
\newlength{\hs@padl}\setlength{\hs@padl}{9pt}
\newlength{\hs@padr}\setlength{\hs@padr}{7pt}
\newlength{\hs@padv}\setlength{\hs@padv}{6pt}
\newcommand{\hs@openbox}{%
  \par\addvspace{1.0\baselineskip}\noindent
  \begin{lrbox}{\hs@cbox}%
  \begin{minipage}{\dimexpr\linewidth-\hs@rule-\hs@padl-\hs@padr\relax}%
  \vspace{\hs@padv}\small\raggedright\setlength{\parskip}{0.35\baselineskip}\relax}
\newcommand{\hs@closebox}[2]{%
  \vspace{\hs@padv}\end{minipage}\end{lrbox}%
  {\setlength{\fboxsep}{0pt}\colorbox{#2}{%
     \textcolor{#1}{\rule[-\dp\hs@cbox]{\hs@rule}{\dimexpr\ht\hs@cbox+\dp\hs@cbox\relax}}%
     \hspace{\hs@padl}\usebox{\hs@cbox}\hspace{\hs@padr}}}%
  \par\addvspace{0.75\baselineskip}}
\newenvironment{insight}{\hs@openbox}{\hs@closebox{accent}{pale}}
\newenvironment{keybox}{\hs@openbox}{\hs@closebox{black!55}{black!4}}
\makeatother

% Boxed statements in the callout look, with the source unchanged:
% \boxtheorem{proposition} in the preamble, then the usual
% \begin{proposition}[Title] ... \end{proposition}. Delete the
% \boxtheorem line and the output is plain amsthm again.
\newcommand{\boxtheorem}[1]{%
  \expandafter\let\csname hs@orig@#1\expandafter\endcsname\csname #1\endcsname
  \expandafter\let\csname hs@orig@end#1\expandafter\endcsname\csname end#1\endcsname
  \renewenvironment{#1}{\hs@openbox\csname hs@orig@#1\endcsname}%
    {\csname hs@orig@end#1\endcsname\hs@closebox{accent}{pale}}}
\makeatother
