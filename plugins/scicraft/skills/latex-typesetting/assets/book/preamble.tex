% =====================================================================
%  preamble.tex: a free-form book or long report (monograph,
%  technical report) in the classic manner of Bringhurst's book design,
%  rebuilt on A4 from common packages (no style package, no placeins,
%  no epigraph package).
%  pdfLaTeX + BibTeX (natbib). Copy shims.tex next to this file.
%
%  The look, element by element:
%   - Palatino text and matching Palatino math, true small caps and
%     old-style figures (mathpazo sc,osf), leading 1.06
%   - chapter opener: a large navy numeral flush right, the title in
%     letterspaced capitals, a full-width hairline, an optional epigraph
%   - section heads in letterspaced lowercase small caps, subsection
%     heads in italic, nothing set heavier than the level above it
%   - running heads in letterspaced small caps, page number at the outer
%     edge, no head rule, never wrapping to a second line
%   - captions: bold label, bold claim sentence, small type
%   - ruled algorithm floats, booktabs tables with the caption on top
%   - one link color for citations and cross-references
%  Measure 150 mm, symmetric margins, no sidenote column: a sidenote
%  column cost a centring fix, broke float placement, and was dropped.
%  When you copy a reference document's look, measure its PDF, and write
%  down here what was copied and where you deviate.
% =====================================================================

% Part pages carry nothing else and are followed by a blank verso.
\makeatletter
\renewcommand\part{%
  \cleardoublepage
  \thispagestyle{empty}%
  \if@twocolumn\onecolumn\@tempswatrue\else\@tempswafalse\fi
  \null\vfil
  \secdef\@part\@spart}
\def\@part[#1]#2{%
  \ifnum \c@secnumdepth >-2\relax
    \refstepcounter{part}%
    \addcontentsline{toc}{part}{\protect\numberline{\thepart}#1}%
  \else
    \addcontentsline{toc}{part}{#1}%
  \fi
  \markboth{}{}%
  {\centering\interlinepenalty\@M\normalfont
   {\normalsize Part \thepart\par}\vskip 5mm
   {\color{navy}\large\spacedallcaps{#2}\par}}%
  \@endpart}
\def\@endpart{\vfil\newpage
  \if@twoside\if@openright\null\thispagestyle{empty}\newpage\fi\fi
  \if@tempswa\twocolumn\fi}
% A verso left empty by \cleardoublepage carries no running head.
\def\cleardoublepage{\clearpage\if@twoside\ifodd\c@page\else
  \hbox{}\thispagestyle{empty}\newpage\if@twocolumn\hbox{}\newpage\fi\fi\fi}
\makeatother

% The bracket that opens each research chapter ("Contributions"):
% a thin vertical rule at the left edge of the measure, interrupted by a
% rotated blue label, with the text set 6.5 mm in.
\newenvironment{contributionbox}[1][Contributions]{%
  \def\bookboxlabel{#1}%
  \par\addvspace{2mm}\noindent
  \begin{lrbox}{\bookbox}%
  \begin{minipage}[t]{\dimexpr\linewidth-\bracketindent\relax}\ignorespaces}
 {\end{minipage}\end{lrbox}%
  \begin{tikzpicture}[baseline={(0,0)}]
    \node[inner sep=0pt,outer sep=0pt,anchor=north west] (body) at (\bracketindent,0) {\usebox{\bookbox}};
    \coordinate (top) at (\bracketrulex,0);
    \path let \p1=(body.south) in coordinate (bot) at (\bracketrulex,{min(\y1,-24mm)});
    \node[overlay,rotate=90,inner sep=1.6pt,font=\scriptsize\color{linkblue}] (lab) at ($(top)!0.5!(bot)$) {\bookboxlabel};
    \draw[line width=0.55pt,rulegray] ([yshift=0.4mm]top) -- (lab.east);
    \draw[line width=0.55pt,rulegray] (lab.west) -- ([yshift=-0.6mm]bot);
  \end{tikzpicture}\par\addvspace{8mm}}
\newenvironment{contribution}{\begin{contributionbox}}{\end{contributionbox}}

% Context boxes \begin{infobox}{box:label}{Title}: a framed panel numbered per chapter
% ("Box 1.2"), for background a reader can skip. A box cannot break across pages: keep it short.
\newcounter{infobox}[chapter]
\renewcommand{\theinfobox}{\thechapter.\arabic{infobox}}
\newenvironment{infobox}[2]{%
  \par\addvspace{10pt plus 2pt}\noindent
  \refstepcounter{infobox}\label{#1}%
  \begin{lrbox}{\bookbox}%
  \begin{minipage}{\dimexpr\linewidth-7.2mm\relax}\small
  {\color{navy}\spacedlowsmallcaps{Box \theinfobox}}\hspace{0.7em}\textbf{#2}\par\smallskip\ignorespaces}
 {\end{minipage}\end{lrbox}%
  \begin{tikzpicture}[baseline={(0,0)},trim left=0pt,trim right=\linewidth]
    \node[inner sep=0pt,anchor=north west] (body) at (3.6mm,-2.6mm) {\usebox{\bookbox}};
    \path (0,0 |- body.south) ++(0,-2.6mm) coordinate (bl);
    \draw[rulegray,line width=0.45pt] (0,0) rectangle (\linewidth,0 |- bl);
    \fill[navy] (0,0) rectangle (0.39mm,0 |- bl);
  \end{tikzpicture}\par\addvspace{10pt plus 2pt}}
